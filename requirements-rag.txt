# Optional dependency for the RAG embedding layer (ai/rag/embeddings.py).
#
# Deliberately kept out of requirements.txt.  sentence-transformers pulls in
# PyTorch, which is roughly 2 GB installed -- an unreasonable cost for anyone
# who only wants the DPI engine or the LLM analysis layer, both of which run
# without it.  Steps 1 and 2 of the RAG layer (the knowledge corpus and the
# chunker) also need nothing from this file.
#
# Install when you are ready to embed:
#
#     pip install -r requirements-rag.txt
#
# First use downloads BAAI/bge-small-en-v1.5 (~130 MB) into the local model
# cache.  After that, embedding is fully offline: no API key, no provider, no
# network.  Set DPI_EMBED_OFFLINE=1 to require the cached copy and fail rather
# than fetch.

sentence-transformers>=3.0
